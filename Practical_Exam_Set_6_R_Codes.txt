QUESTION 2: PERFORM OLS REGRESSION AND EXAMINE RESIDUALS


Step 1: Create data


X <- c(1, 2, 3, 4, 5, 6, 7, 8) Y <- c(3, 5, 8, 9, 12, 13, 16, 18)


data <- data.frame(X, Y) print(data)


Step 2: Fit simple linear regression model


model <- lm(Y ~ X, data = data)


print(“Model Summary”) summary(model)


Step 3: Calculate fitted values


fitted_values <- fitted(model)


print(“Fitted Values”) print(fitted_values)


Step 4: Calculate residuals


res <- residuals(model)


print(“Residuals”) print(res)


Step 5: Plot residuals against fitted values


plot( fitted_values, res, main = “Residuals vs Fitted Values”, xlab =
“Fitted Values”, ylab = “Residuals”, pch = 19, col = “blue” )


Step 6: Add horizontal reference line at zero


abline(h = 0, col = “red”, lwd = 2)

Install packages only if they are not already installed

if (!requireNamespace(“tidyr”, quietly = TRUE)) {

Step 1: Create data in wide format


Student <- c(“Ana”, “Bilal”, “Chetna”, “Danish”, “Ekta”)


Maths <- c(62, 75, 48, 89, 70)


Statistics <- c(68, 79, 52, 91, 73)


Computer <- c(71, 82, 55, 94, 76)


data <- data.frame( Student, Maths, Statistics, Computer )


print(“Wide Format Data”) print(data)


Step 2: Reshape wide data into long format


long_data <- pivot_longer( data, cols = c(Maths, Statistics, Computer),
names_to = “Subject”, values_to = “Marks” )


print(“Long Format Data”) print(long_data)


Step 3: Create comparative box plots


ggplot( long_data, aes(x = Subject, y = Marks, fill = Subject) ) +
geom_boxplot() + labs( title = “Comparison of Marks in Three Subjects”,
x = “Subjects”, y = “Marks” ) + theme_minimal()
